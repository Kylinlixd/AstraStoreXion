# 星辰离子X 客户端使用文档

## 简介

星辰离子X是一个高性能分布式文件存储系统，提供多种编程语言的客户端SDK，支持文件上传、下载、删除和状态查询等功能。当前支持Go、Java、NodeJS和Python编程语言。

## 安装

### Go 客户端

```bash
go get github.com/astrastore/astrastore-xion/client/go
```

### Java 客户端

在您的Maven项目的`pom.xml`文件中添加以下依赖:

```xml
<dependency>
  <groupId>com.astrastore</groupId>
  <artifactId>astrastore-xion-client</artifactId>
  <version>1.0.0</version>
</dependency>
```

### NodeJS 客户端

```bash
npm install astrastore-xion-client
```

### Python 客户端

```bash
pip install astrastore-xion-client
```

## 配置

所有客户端都支持通过配置文件或代码配置。以下是YAML配置文件示例:

```yaml
api_gateway: "http://localhost:8080"
timeout: 30000
max_retries: 3
retry_interval: 1000
chunk_size: 4194304  # 4MB
parallel_uploads: 3
parallel_downloads: 3
enable_cache: true
cache_ttl: 60000
cache_max_size: 104857600  # 100MB
```

## 使用示例

### Go 客户端

```go
package main

import (
    "context"
    "fmt"
    "io"
    "os"
    
    "github.com/astrastore/astrastore-xion/client/go"
)

func main() {
    // 加载配置
    cfg, err := client.LoadConfig("config.yaml")
    if err != nil {
        fmt.Printf("加载配置失败: %v\n", err)
        return
    }
    
    // 创建客户端
    xionClient, err := client.NewClient(cfg)
    if err != nil {
        fmt.Printf("创建客户端失败: %v\n", err)
        return
    }
    defer xionClient.Close()
    
    ctx := context.Background()
    
    // 上传文件
    file, err := os.Open("example.txt")
    if err != nil {
        fmt.Printf("打开文件失败: %v\n", err)
        return
    }
    defer file.Close()
    
    metadata := map[string]string{
        "content_type": "text/plain",
        "description": "示例文本文件",
    }
    
    uploadResp, err := xionClient.UploadFile(ctx, file, "example.txt", metadata)
    if err != nil {
        fmt.Printf("上传文件失败: %v\n", err)
        return
    }
    
    fmt.Printf("文件上传成功，ID: %s\n", uploadResp.FileID)
    
    // 获取文件状态
    statusResp, err := xionClient.GetFileStatus(ctx, uploadResp.FileID)
    if err != nil {
        fmt.Printf("获取文件状态失败: %v\n", err)
        return
    }
    
    fmt.Printf("文件状态: %s\n", statusResp.Status)
    
    // 下载文件
    outFile, err := os.Create("downloaded-example.txt")
    if err != nil {
        fmt.Printf("创建输出文件失败: %v\n", err)
        return
    }
    defer outFile.Close()
    
    err = xionClient.DownloadFile(ctx, uploadResp.FileID, outFile)
    if err != nil {
        fmt.Printf("下载文件失败: %v\n", err)
        return
    }
    
    fmt.Println("文件下载成功")
    
    // 删除文件
    deleteResp, err := xionClient.DeleteFile(ctx, uploadResp.FileID)
    if err != nil {
        fmt.Printf("删除文件失败: %v\n", err)
        return
    }
    
    fmt.Printf("文件删除结果: %v\n", deleteResp.Success)
}
```

### Java 客户端

```java
import com.astrastore.client.XionClient;
import com.astrastore.client.XionConfig;
import com.astrastore.client.model.*;

import java.io.File;
import java.io.FileInputStream;
import java.io.FileOutputStream;
import java.io.InputStream;
import java.io.OutputStream;
import java.util.HashMap;
import java.util.Map;

public class XionClientExample {
    public static void main(String[] args) {
        try {
            // 创建配置
            XionConfig config = XionConfig.fromYamlFile("config.yaml");
            
            // 创建客户端
            try (XionClient client = new XionClient(config)) {
                // 准备上传
                File file = new File("example.txt");
                InputStream inputStream = new FileInputStream(file);
                Map<String, String> metadata = new HashMap<>();
                metadata.put("content_type", "text/plain");
                metadata.put("description", "示例文本文件");
                
                // 上传文件
                UploadFileResponse uploadResponse = client.uploadFile(inputStream, "example.txt", metadata);
                System.out.println("文件上传成功，ID: " + uploadResponse.getFileId());
                
                // 获取文件状态
                FileStatusResponse statusResponse = client.getFileStatus(uploadResponse.getFileId());
                System.out.println("文件状态: " + statusResponse.getStatus());
                
                // 下载文件
                OutputStream outputStream = new FileOutputStream("downloaded-example.txt");
                client.downloadFile(uploadResponse.getFileId(), outputStream);
                System.out.println("文件下载成功");
                outputStream.close();
                
                // 删除文件
                DeleteFileResponse deleteResponse = client.deleteFile(uploadResponse.getFileId());
                System.out.println("文件删除结果: " + deleteResponse.isSuccess());
            }
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
```

### NodeJS 客户端

```javascript
const fs = require('fs');
const { XionClient, loadConfigFromYaml } = require('astrastore-xion-client');

async function main() {
  try {
    // 加载配置
    const config = loadConfigFromYaml('config.yaml');
    
    // 创建客户端
    const client = new XionClient(config);
    
    // 上传文件
    const fileStream = fs.createReadStream('example.txt');
    const metadata = {
      content_type: 'text/plain',
      description: '示例文本文件'
    };
    
    const uploadResponse = await client.uploadFile(fileStream, 'example.txt', metadata);
    console.log(`文件上传成功，ID: ${uploadResponse.fileId}`);
    
    // 获取文件状态
    const statusResponse = await client.getFileStatus(uploadResponse.fileId);
    console.log(`文件状态: ${statusResponse.status}`);
    
    // 下载文件
    const downloadResponse = await client.downloadFile(uploadResponse.fileId);
    const outputStream = fs.createWriteStream('downloaded-example.txt');
    downloadResponse.stream.pipe(outputStream);
    
    await new Promise((resolve, reject) => {
      outputStream.on('finish', resolve);
      outputStream.on('error', reject);
    });
    
    console.log('文件下载成功');
    
    // 删除文件
    const deleteResponse = await client.deleteFile(uploadResponse.fileId);
    console.log(`文件删除结果: ${deleteResponse.success}`);
  } catch (error) {
    console.error('发生错误:', error);
  }
}

main();
```

### Python 客户端

```python
from astrastore_xion import XionClient, load_config_from_yaml

def main():
    try:
        # 加载配置
        config = load_config_from_yaml('config.yaml')
        
        # 创建客户端
        client = XionClient(config)
        
        # 上传文件
        with open('example.txt', 'rb') as file:
            metadata = {
                'content_type': 'text/plain',
                'description': '示例文本文件'
            }
            upload_response = client.upload_file(file, 'example.txt', metadata)
        
        print(f"文件上传成功，ID: {upload_response.file_id}")
        
        # 获取文件状态
        status_response = client.get_file_status(upload_response.file_id)
        print(f"文件状态: {status_response.status}")
        
        # 下载文件
        with open('downloaded-example.txt', 'wb') as output_file:
            client.download_file(upload_response.file_id, output_file)
        
        print("文件下载成功")
        
        # 删除文件
        delete_response = client.delete_file(upload_response.file_id)
        print(f"文件删除结果: {delete_response.success}")
    except Exception as e:
        print(f"发生错误: {e}")

if __name__ == "__main__":
    main()
```

## API参考

### 上传文件

上传一个文件到存储系统。

**参数:**
- `file`: 文件内容（文件对象、流或字节数组）
- `filename`: 文件名
- `metadata`: 可选，文件元数据，键值对

**返回:**
- `file_id`: 文件ID
- `success`: 是否成功
- `message`: 消息

### 下载文件

下载指定ID的文件。

**参数:**
- `file_id`: 文件ID
- `output`: 输出目标（文件对象、流）

**返回:**
- 根据客户端不同，可能返回流或直接写入指定输出

### 删除文件

删除指定ID的文件。

**参数:**
- `file_id`: 文件ID

**返回:**
- `success`: 是否成功
- `message`: 消息

### 获取文件状态

获取指定ID文件的状态信息。

**参数:**
- `file_id`: 文件ID

**返回:**
- `file_id`: 文件ID
- `filename`: 文件名
- `size`: 文件大小（字节）
- `status`: 文件状态（"available", "pending", "corrupted"）
- `metadata`: 文件元数据
- `chunks`: 文件块信息（可选）

## 错误处理

所有客户端都提供了一致的错误处理机制。常见错误包括:

- 连接错误: 无法连接到API网关
- 超时错误: 请求超时
- 认证错误: 认证失败
- 权限错误: 权限不足
- 资源错误: 资源不存在或已删除
- 服务器错误: 服务器内部错误

## 配置参考

| 参数 | 说明 | 默认值 |
|------|------|--------|
| api_gateway | API网关地址 | http://localhost:8080 |
| timeout | 请求超时时间（毫秒） | 30000 |
| max_retries | 最大重试次数 | 3 |
| retry_interval | 重试间隔（毫秒） | 1000 |
| chunk_size | 分块大小（字节） | 4194304 (4MB) |
| parallel_uploads | 并行上传数 | 3 |
| parallel_downloads | 并行下载数 | 3 |
| enable_cache | 是否启用缓存 | true |
| cache_ttl | 缓存有效期（毫秒） | 60000 |
| cache_max_size | 缓存最大大小（字节） | 104857600 (100MB) |
